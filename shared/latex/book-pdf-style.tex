% KOMA typography
\KOMAoptions{
  headings=small,
  chapterprefix=false,
  numbers=noenddot,
  headinclude=true,
  footinclude=false
}

\setkomafont{disposition}{\sffamily\bfseries\color{BookInk}}

\setkomafont{section}{\Large}
